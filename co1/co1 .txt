#include <stdio.h>
#include <limits.h>
 
typedef unsigned short meter_t;          /* 16-bit meter counter */
typedef enum { DOMESTIC, COMMERCIAL, INDUSTRIAL } Category;
 
#define SLAB1_UNITS        100
#define SLAB2_UNITS        200
#define RATE2_PAISE        225LL
#define RATE3_PAISE        450LL
#define FIXED_CHARGE_PAISE 5000LL
 
typedef struct {
    meter_t previous;          /* last meter value            */
    meter_t current;           /* latest meter value          */
    char household[11];        /* text: keeps leading zeroes  */
    Category category;         /* small fixed set of labels   */
    unsigned int units;        /* difference incl. rollover   */
    long long energy_paise;    /* slab charge                 */
    long long fixed_paise;     /* fixed charge                */
    long long total_paise;     /* energy + fixed              */
} Bill;
 
static const char *cat_name[] = {"Domestic", "Commercial", "Industrial"};
 
long long energy_charge(unsigned int u)
{
    long long e = 0;
    if (u > SLAB2_UNITS) e += (long long)(u - SLAB2_UNITS) * RATE3_PAISE;
    if (u > SLAB1_UNITS)
        e += (long long)((u < SLAB2_UNITS ? u : SLAB2_UNITS) - SLAB1_UNITS) * RATE2_PAISE;
    return e;                  /* first 100 units are free */
}
 
void show(const Bill *b)
{
    printf("Household number : %s\n", b->household);
    printf("Category         : %s\n", cat_name[b->category]);
    printf("Previous reading : %u\n", (unsigned)b->previous);
    printf("Current reading  : %u\n", (unsigned)b->current);
    printf("Units consumed   : %u\n", b->units);
    printf("Energy charge    : Rs %lld.%02lld\n", b->energy_paise / 100, b->energy_paise % 100);
    printf("Fixed charge     : Rs %lld.%02lld\n", b->fixed_paise / 100, b->fixed_paise % 100);
    printf("Total            : Rs %lld.%02lld\n", b->total_paise / 100, b->total_paise % 100);
}
 
int main(void)
{
    printf("---- Data items, types and ranges ----\n");
    printf("%-18s %-20s %zu byte(s)  range 0..%u\n", "Previous/Current", "unsigned short", sizeof(meter_t), (unsigned)USHRT_MAX);
    printf("%-18s %-20s %zu byte(s)  range 0..%u\n", "Units consumed", "unsigned int", sizeof(unsigned int), UINT_MAX);
    printf("%-18s %-20s %zu byte(s)  (text, 10 chars + NUL)\n", "Household number", "char[11]", sizeof(char[11]));
    printf("%-18s %-20s %zu byte(s)  values 0..2\n", "Category", "enum", sizeof(Category));
    printf("%-18s %-20s %zu byte(s)  max %lld\n\n", "Money (paise)", "long long", sizeof(long long), LLONG_MAX);
 
    Bill b = { 65400, 120, "0000012345", DOMESTIC, 0, 0, FIXED_CHARGE_PAISE, 0 };
 
    b.units        = (unsigned short)(b.current - b.previous);   /* rollover */
    b.energy_paise = energy_charge(b.units);
    b.total_paise  = b.energy_paise + b.fixed_paise;
 
    printf("---- Sample bill (meter rolled over) ----\n");
    show(&b);
    return 0;
}

#include <stdio.h>
 
#define SLAB1_UNITS 100
#define SLAB2_UNITS 200
#define RATE1_PAISE 0
#define RATE2_PAISE 225
#define RATE3_PAISE 450
#define FIXED_CHARGE_PAISE 5000L
 
long compute_bill(int units)
{
    long paise = FIXED_CHARGE_PAISE;
 
    if (units > SLAB2_UNITS)
        paise += (long)(units - SLAB2_UNITS) * RATE3_PAISE;
    if (units > SLAB1_UNITS) {
        int slab2 = units < SLAB2_UNITS ? units - SLAB1_UNITS
                                        : SLAB2_UNITS - SLAB1_UNITS;
        paise += (long)slab2 * RATE2_PAISE;
    }
    /* First slab rate is zero, but included for clarity. */
    if (units > 0 && units <= SLAB1_UNITS)
        paise += (long)units * RATE1_PAISE;
    return paise;
}
 
int main(void)
{
    int tests[] = {0, 100, 150, 200, 350};
    int n = sizeof tests / sizeof tests[0];
 
    for (int i = 0; i < n; ++i) {
        long p = compute_bill(tests[i]);
        printf("%3d units -> Rs %ld.%02ld\n", tests[i], p / 100, p % 100);
    }
    return 0;
}

#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
 
typedef unsigned short meter_t;
 
#define SLAB1_UNITS 100
#define SLAB2_UNITS 200
#define RATE1_PAISE 0L
#define RATE2_PAISE 225L
#define RATE3_PAISE 450L
#define FIXED_CHARGE_PAISE 5000L
 
unsigned int consumption(meter_t previous, meter_t current)
{
    return (unsigned short)(current - previous);
}
 
long long compute_bill(unsigned int units)
{
    long long energy = 0;
 
    if (units > SLAB2_UNITS)
        energy += (long long)(units - SLAB2_UNITS) * RATE3_PAISE;
    if (units > SLAB1_UNITS) {
        unsigned int middle = units < SLAB2_UNITS
                              ? units - SLAB1_UNITS
                              : SLAB2_UNITS - SLAB1_UNITS;
        energy += (long long)middle * RATE2_PAISE;
    }
    if (units <= SLAB1_UNITS)
        energy += (long long)units * RATE1_PAISE;
    return energy + FIXED_CHARGE_PAISE;
}
 
int main(void)
{
    unsigned int previous, current, units;
 
    printf("Enter previous and current meter readings: ");
    if (scanf("%u %u", &previous, &current) != 2 ||
        previous > USHRT_MAX || current > USHRT_MAX) {
        printf("Invalid meter input\n");
        return 1;
    }
 
    units = consumption((meter_t)previous, (meter_t)current);
    if (current < previous && units > 10000) {
        printf("Reading difference requires manual rollover validation\n");
        return 1;
    }
 
    long long total = compute_bill(units);
    printf("Units: %u\n", units);
    printf("Energy and fixed total: Rs %lld.%02lld\n",
           total / 100, total % 100);
    return 0;
}

// Calculate units with single-rollover handling
unsigned int calculate_units(unsigned short prev, unsigned short curr) {
    if (curr >= prev) {
        return (unsigned int)(curr - prev);
    } else {
        return (unsigned int)((METER_MAX - prev) + curr + 1);
    }
}

// Compute energy charge using exact integer paise arithmetic
long long compute_energy_charge(unsigned int units) {
    long long paise = 0;
    if (units <= SLAB1_UNITS) {
        paise = (long long)units * RATE1_PAISE;
    } else if (units <= SLAB2_UNITS) {
        paise = (SLAB1_UNITS * RATE1_PAISE) + ((long long)(units - SLAB1_UNITS) * RATE2_PAISE);
    } else {
        paise = (SLAB1_UNITS * RATE1_PAISE) + 
                ((SLAB2_UNITS - SLAB1_UNITS) * RATE2_PAISE) + 
                ((long long)(units - SLAB2_UNITS) * RATE3_PAISE);
    }
    return paise;
}

// Robust input validation using fgets and strtol
int get_valid_reading(const char *prompt, unsigned short *reading) {
    char buffer[100];
    char *endptr;
    long val;

    printf("%s", prompt);
    if (!fgets(buffer, sizeof(buffer), stdin)) {
        return 0;
    }

    // Remove trailing newline if present
    buffer[strcspn(buffer, "\n")] = 0;

    val = strtol(buffer, &endptr, 10);
    if (*endptr != '\0' || val < 0 || val > 65535) {
        return 0; // Invalid input or out of 16-bit range
    }

    *reading = (unsigned short)val;
    return 1;
}

#include <stdio.h>
 
typedef unsigned long long energy_t;
typedef long long money_t;
 
typedef struct {
    money_t base_rate_paise_per_kwh;
    money_t fixed_charge_paise;
} Tariff;
 
money_t charge_for_wh(energy_t wh, money_t rate_paise_per_kwh)
{
    /* paise = Wh * paise/kWh / 1000; nearest-paise rounding */
    return (money_t)((wh * (energy_t)rate_paise_per_kwh + 500) / 1000);
}
 
int main(void)
{
    Tariff tariff;
    energy_t peak_wh, offpeak_wh;
    money_t peak_rate, offpeak_charge, peak_charge, total;
 
    printf("Enter base rate (paise/kWh) and fixed charge (paise): ");
    if (scanf("%lld %lld", &tariff.base_rate_paise_per_kwh,
              &tariff.fixed_charge_paise) != 2 ||
        tariff.base_rate_paise_per_kwh < 0 ||
        tariff.fixed_charge_paise < 0) {
        printf("Invalid tariff\n");
        return 1;
    }
 
    printf("Enter peak Wh and off-peak Wh: ");
    if (scanf("%llu %llu", &peak_wh, &offpeak_wh) != 2) {
        printf("Invalid energy input\n");
        return 1;
    }
 
    peak_rate = (tariff.base_rate_paise_per_kwh * 120 + 50) / 100;
    peak_charge = charge_for_wh(peak_wh, peak_rate);
    offpeak_charge = charge_for_wh(offpeak_wh, tariff.base_rate_paise_per_kwh);
    total = peak_charge + offpeak_charge + tariff.fixed_charge_paise;
 
    printf("Peak charge: Rs %lld.%02lld\n", peak_charge / 100, peak_charge % 100);
    printf("Off-peak charge: Rs %lld.%02lld\n", offpeak_charge / 100, offpeak_charge % 100);
    printf("Total: Rs %lld.%02lld\n", total / 100, total % 100);
    return 0;
}
